"""
كود شامل لتدريب نموذج YOLO11
يشمل: إنشاء المجلدات، تقسيم البيانات، إنشاء classes.txt، إنشاء data.yaml، التدريب، وحفظ النموذج
"""

from ultralytics import YOLO
import torch
import os
import yaml
import shutil
from datetime import datetime
import random
from pathlib import Path

# ==================== الإعدادات الرئيسية ====================
base_path = r'C:\Users\z\Desktop\New folder (3)'

# أسماء الفئات - عدّلها حسب مشروعك
class_names = [
    'person',
    'car',
    'bicycle',
    'dog',
    'cat'
]

# إعدادات التقسيم
TRAIN_PERCENTAGE = 0.8  # 80% للتدريب، 20% للـ validation
RANDOM_SEED = 42

# إعدادات التدريب
EPOCHS = 60
IMG_SIZE = 640
MODEL_NAME = 'yolo11s.pt'

# ==================== 1. إنشاء هيكل المجلدات ====================
print("="*70)
print("STEP 1: Creating Folder Structure")
print("="*70)

folders = [
    'custom_data',
    'train_split/images',  # مجلدات التدريب بعد التقسيم
    'train_split/labels',
    'validation/images',
    'validation/labels',
    'test/images',
    'test/labels'
]

for folder in folders:
    folder_path = os.path.join(base_path, folder)
    os.makedirs(folder_path, exist_ok=True)
    print(f"✓ Created/Verified: {folder}")

print(f"\n✓ All folders created successfully!")

# ==================== 2. تقسيم البيانات ====================
print("\n" + "="*70)
print("STEP 2: Splitting Dataset (Train/Validation)")
print("="*70)

# قراءة من مجلد train الموجود
dataset_images_path = os.path.join(base_path, 'train', 'images')
dataset_labels_path = os.path.join(base_path, 'train', 'labels')

# البحث عن الصور في المجلد الأصلي
all_images = [f for f in os.listdir(dataset_images_path) 
              if f.lower().endswith(('.jpg', '.jpeg', '.png', '.bmp'))]

if len(all_images) > 0:
    print(f"📊 Found {len(all_images)} images in dataset folder")
    print(f"📐 Split ratio: {TRAIN_PERCENTAGE*100}% train, {(1-TRAIN_PERCENTAGE)*100}% validation")
    
    # خلط البيانات بشكل عشوائي
    random.seed(RANDOM_SEED)
    random.shuffle(all_images)
    
    # حساب عدد الصور للتدريب
    train_count = int(len(all_images) * TRAIN_PERCENTAGE)
    train_images = all_images[:train_count]
    val_images = all_images[train_count:]
    
    print(f"\n📦 Training images: {len(train_images)}")
    print(f"📦 Validation images: {len(val_images)}")
    
    # نسخ ملفات التدريب
    print("\n🔄 Copying training files...")
    train_copied = 0
    train_labels_found = 0
    
    for img_name in train_images:
        # نسخ الصورة
        src_img = os.path.join(dataset_images_path, img_name)
        dst_img = os.path.join(base_path, 'train_split', 'images', img_name)
        shutil.copy2(src_img, dst_img)
        train_copied += 1
        
        # نسخ الـ label المقابل
        label_name = os.path.splitext(img_name)[0] + '.txt'
        src_label = os.path.join(dataset_labels_path, label_name)
        dst_label = os.path.join(base_path, 'train_split', 'labels', label_name)
        
        if os.path.exists(src_label):
            shutil.copy2(src_label, dst_label)
            train_labels_found += 1
    
    print(f"✓ Copied {train_copied} training images")
    print(f"✓ Found {train_labels_found} training labels")
    
    # نسخ ملفات الـ validation
    print("\n🔄 Copying validation files...")
    val_copied = 0
    val_labels_found = 0
    
    for img_name in val_images:
        # نسخ الصورة
        src_img = os.path.join(dataset_images_path, img_name)
        dst_img = os.path.join(base_path, 'validation', 'images', img_name)
        shutil.copy2(src_img, dst_img)
        val_copied += 1
        
        # نسخ الـ label المقابل
        label_name = os.path.splitext(img_name)[0] + '.txt'
        src_label = os.path.join(dataset_labels_path, label_name)
        dst_label = os.path.join(base_path, 'validation', 'labels', label_name)
        
        if os.path.exists(src_label):
            shutil.copy2(src_label, dst_label)
            val_labels_found += 1
    
    print(f"✓ Copied {val_copied} validation images")
    print(f"✓ Found {val_labels_found} validation labels")
    
    # تحذير إذا كانت بعض الـ labels مفقودة
    if train_labels_found < len(train_images):
        print(f"\n⚠️  Warning: {len(train_images) - train_labels_found} training labels missing!")
    if val_labels_found < len(val_images):
        print(f"⚠️  Warning: {len(val_images) - val_labels_found} validation labels missing!")
    
    print(f"\n✅ Dataset split completed successfully!")

else:
    print(f"⚠️  No images found in {dataset_images_path}")
    print(f"\n📝 Current folder structure check:")
    print(f"  Looking for images in: {dataset_images_path}")
    if os.path.exists(dataset_images_path):
        all_files = os.listdir(dataset_images_path)
        print(f"  Files found: {len(all_files)}")
        if len(all_files) > 0:
            print(f"  First few files: {all_files[:5]}")
    else:
        print(f"  ❌ Folder doesn't exist!")
    
    print(f"\n📁 Please make sure your images are in:")
    print(f"  {dataset_images_path}")
    print(f"  And labels are in:")
    print(f"  {dataset_labels_path}")
    
    print(f"\n💡 Skipping dataset split. Will check train_split/ and validation/ folders...")
    
    # في حال عدم وجود صور للتقسيم، نستخدم المجلدات الموجودة
    print(f"\n⚠️  Will attempt to use existing train_split/ and validation/ folders if available")

# ==================== 3. إنشاء ملف classes.txt ====================
print("\n" + "="*70)
print("STEP 3: Creating classes.txt")
print("="*70)

classes_txt_path = os.path.join(base_path, 'custom_data', 'classes.txt')

with open(classes_txt_path, 'w', encoding='utf-8') as f:
    for class_name in class_names:
        f.write(class_name.strip() + '\n')

print(f"✓ Created: {classes_txt_path}")
print(f"  Number of classes: {len(class_names)}")
print(f"  Classes: {class_names}")

# ==================== 4. إنشاء ملف data.yaml ====================
print("\n" + "="*70)
print("STEP 4: Creating data.yaml")
print("="*70)

data_yaml_path = os.path.join(base_path, 'data.yaml')

data = {
    'path': base_path,
    'train': 'train_split/images',
    'val': 'validation/images',
    'test': 'test/images',
    'nc': len(class_names),
    'names': class_names
}

with open(data_yaml_path, 'w', encoding='utf-8') as f:
    yaml.dump(data, f, sort_keys=False, allow_unicode=True)

print(f"✓ Created: {data_yaml_path}")
print("\ndata.yaml contents:")
print("-"*70)
with open(data_yaml_path, 'r', encoding='utf-8') as f:
    print(f.read())
print("-"*70)

# ==================== 5. التحقق من البيانات النهائية ====================
print("\n" + "="*70)
print("STEP 5: Final Dataset Verification")
print("="*70)

train_images_path = os.path.join(base_path, 'train_split', 'images')
train_labels_path = os.path.join(base_path, 'train_split', 'labels')
val_images_path = os.path.join(base_path, 'validation', 'images')
val_labels_path = os.path.join(base_path, 'validation', 'labels')

# عد الصور والـ labels
train_images = [f for f in os.listdir(train_images_path) if f.lower().endswith(('.jpg', '.jpeg', '.png', '.bmp'))]
val_images = [f for f in os.listdir(val_images_path) if f.lower().endswith(('.jpg', '.jpeg', '.png', '.bmp'))]
train_labels = [f for f in os.listdir(train_labels_path) if f.endswith('.txt')]
val_labels = [f for f in os.listdir(val_labels_path) if f.endswith('.txt')]

print(f"📊 Final Dataset Statistics:")
print(f"  Training images: {len(train_images)}")
print(f"  Training labels: {len(train_labels)}")
print(f"  Validation images: {len(val_images)}")
print(f"  Validation labels: {len(val_labels)}")
print(f"  Total images: {len(train_images) + len(val_images)}")

# تحذير إذا لم توجد بيانات
if len(train_images) == 0 or len(val_images) == 0:
    print("\n" + "!"*70)
    print("⚠️  WARNING: Insufficient data for training!")
    print("!"*70)
    print("\n📁 Please add your data to:")
    print(f"  {dataset_images_path}")
    print(f"  {dataset_labels_path}")
    print(f"\n📝 Label format (one line per object):")
    print(f"    class_id center_x center_y width height")
    print(f"    Example: 0 0.5 0.5 0.3 0.4")
    print(f"\n💡 All coordinates should be normalized (0-1)")
    print("\n" + "!"*70)
    
    response = input("\n❓ Do you want to continue anyway for testing? (yes/no): ").lower()
    if response != 'yes':
        print("\n❌ Exiting... Please add your dataset first.")
        exit()
    else:
        print("\n⚠️  Continuing without sufficient data (may cause errors)...")

# ==================== 6. معلومات النظام ====================
print("\n" + "="*70)
print("STEP 6: System Information")
print("="*70)

print(f"PyTorch version: {torch.__version__}")
print(f"CUDA available: {torch.cuda.is_available()}")

if torch.cuda.is_available():
    print(f"GPU: {torch.cuda.get_device_name(0)}")
    print(f"CUDA version: {torch.version.cuda}")
    device = 0
    batch_size = 16
    workers = 8
    print("\n✓ Training will use GPU (Fast)")
else:
    device = 'cpu'
    batch_size = 4
    workers = 2
    print("\n⚠️  Training will use CPU (Slow)")
    print("💡 Tip: Use Google Colab for free GPU training")

# ==================== 7. تحميل وتدريب النموذج ====================
print("\n" + "="*70)
print("STEP 7: Loading Model and Starting Training")
print("="*70)

print(f"\n📦 Loading model: {MODEL_NAME}")
model = YOLO(MODEL_NAME)

print(f"\n🎯 Training Configuration:")
print(f"  Device: {device}")
print(f"  Epochs: {EPOCHS}")
print(f"  Image Size: {IMG_SIZE}")
print(f"  Batch Size: {batch_size}")
print(f"  Workers: {workers}")
print("\n" + "="*70)

# بدء التدريب
try:
    start_time = datetime.now()
    print(f"🚀 Training started at: {start_time.strftime('%Y-%m-%d %H:%M:%S')}")
    print("="*70 + "\n")
    
    results = model.train(
        data=data_yaml_path,
        epochs=EPOCHS,
        imgsz=IMG_SIZE,
        device=device,
        batch=batch_size,
        workers=workers,
        project='runs/detect',
        name='yolo11s_train',
        patience=10,
        save=True,
        save_period=10,
        plots=True,
        verbose=True,
        cache=False,
        amp=True if device != 'cpu' else False,
        optimizer='auto',
        lr0=0.01,
        lrf=0.01,
        momentum=0.937,
        weight_decay=0.0005,
        warmup_epochs=3.0,
        box=7.5,
        cls=0.5,
        dfl=1.5,
    )
    
    end_time = datetime.now()
    duration = end_time - start_time
    
    print("\n" + "="*70)
    print("✅ TRAINING COMPLETED SUCCESSFULLY!")
    print("="*70)
    print(f"⏱️  Training duration: {duration}")
    print(f"📊 Best model: {model.trainer.best}")
    print(f"📁 Results saved in: runs/detect/yolo11s_train")
    
    # ==================== 8. حفظ النموذج النهائي ====================
    print("\n" + "="*70)
    print("STEP 8: Saving Final Model")
    print("="*70)
    
    # إنشاء مجلد للنماذج
    models_folder = os.path.join(base_path, 'models')
    os.makedirs(models_folder, exist_ok=True)
    
    # نسخ أفضل نموذج
    best_model_path = model.trainer.best
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    final_model_path = os.path.join(models_folder, f'best_yolo11s_{timestamp}.pt')
    
    if os.path.exists(best_model_path):
        shutil.copy2(best_model_path, final_model_path)
        print(f"✓ Best model copied to: {final_model_path}")
        
        # حفظ أيضاً نسخة باسم ثابت للاستخدام السهل
        simple_path = os.path.join(models_folder, 'best_model.pt')
        shutil.copy2(best_model_path, simple_path)
        print(f"✓ Best model also saved as: {simple_path}")
    
    # حفظ آخر نموذج
    last_model_path = os.path.join(model.trainer.save_dir, 'weights', 'last.pt')
    if os.path.exists(last_model_path):
        final_last_path = os.path.join(models_folder, f'last_yolo11s_{timestamp}.pt')
        shutil.copy2(last_model_path, final_last_path)
        print(f"✓ Last model copied to: {final_last_path}")
    
    # ==================== 9. ملخص النتائج ====================
    print("\n" + "="*70)
    print("STEP 9: Training Summary")
    print("="*70)
    
    # قراءة ملف النتائج
    results_file = os.path.join(model.trainer.save_dir, 'results.csv')
    if os.path.exists(results_file):
        print(f"✓ Results CSV: {results_file}")
    
    # عرض مسارات المخططات
    plots_folder = model.trainer.save_dir
    print(f"\n📊 Training plots saved in: {plots_folder}")
    print(f"  - confusion_matrix.png")
    print(f"  - results.png")
    print(f"  - val_batch predictions")
    
    print("\n" + "="*70)
    print("🎉 ALL STEPS COMPLETED SUCCESSFULLY!")
    print("="*70)
    
    print("\n📝 Summary:")
    print(f"  ✓ Dataset split: {len(train_images)} train, {len(val_images)} validation")
    print(f"  ✓ Training completed in: {duration}")
    print(f"  ✓ Best model saved: {simple_path}")
    
    print("\n💡 To use your trained model:")
    print(f"  from ultralytics import YOLO")
    print(f"  model = YOLO('{simple_path}')")
    print(f"  results = model.predict('image.jpg')")
    print(f"  results[0].show()  # عرض النتائج")

except KeyboardInterrupt:
    print("\n\n⚠️  Training interrupted by user!")
    print("Partial results may be saved in runs/detect/")

except Exception as e:
    print("\n" + "="*70)
    print("❌ ERROR DURING TRAINING")
    print("="*70)
    print(f"Error: {e}")
    print("\n🔧 Possible solutions:")
    print("  1. Make sure you have images in dataset/images folder")
    print("  2. Make sure you have corresponding .txt label files in dataset/labels")
    print("  3. Check that image and label filenames match exactly")
    print("  4. Reduce batch size if getting memory errors")
    print("  5. Check label format is correct")
    print("\n📖 Label format example:")
    print("  For image 'photo.jpg', create 'photo.txt' with:")
    print("  0 0.5 0.5 0.3 0.4  (class_id x_center y_center width height)")
    print("  All values should be normalized (0-1)")
    
    import traceback
    print("\n📋 Full error details:")
    print(traceback.format_exc())

print("\n" + "="*70)
print("Script finished!")
print("="*70)
