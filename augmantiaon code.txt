import os
import cv2
import numpy as np
from tqdm import tqdm

# 📂 المجلد الذي يحتوي على الصور
input_dir = r"C:\Users\z\Desktop\New folder (5)"
output_dir = os.path.join(input_dir, "augmented")
os.makedirs(output_dir, exist_ok=True)

# 🧮 عدد النسخ الجديدة لكل صورة
N = 3  # يمكن تغييره حسب الحاجة

# 🔹 دالة لتعديل السطوع والتباين
def adjust_brightness_contrast(image):
    alpha = np.random.uniform(0.8, 1.2)  # contrast
    beta = np.random.randint(-30, 30)     # brightness
    return cv2.convertScaleAbs(image, alpha=alpha, beta=beta)

# 🔹 دالة لتدوير الصورة ±15 درجة
def rotate_image(image):
    angle = np.random.uniform(-15, 15)
    h, w = image.shape[:2]
    M = cv2.getRotationMatrix2D((w//2, h//2), angle, 1.0)
    return cv2.warpAffine(image, M, (w, h), borderMode=cv2.BORDER_REFLECT)

# 🚀 تنفيذ Augmentation
for img_name in tqdm(os.listdir(input_dir)):
    if not img_name.lower().endswith(('.jpg', '.jpeg', '.png')):
        continue
    
    img_path = os.path.join(input_dir, img_name)
    image = cv2.imread(img_path)
    if image is None:
        continue

    for i in range(N):
        aug_img = image.copy()

        # Flip أفقي أو عمودي
        if np.random.rand() < 0.5:
            aug_img = cv2.flip(aug_img, 1)  # أفقي
        if np.random.rand() < 0.3:
            aug_img = cv2.flip(aug_img, 0)  # عمودي

        # تعديل السطوع والتباين
        aug_img = adjust_brightness_contrast(aug_img)

        # تدوير الصورة
        aug_img = rotate_image(aug_img)

        # حفظ الصورة
        new_name = f"{os.path.splitext(img_name)[0]}_aug_{i}.jpg"
        cv2.imwrite(os.path.join(output_dir, new_name), aug_img)

print("✅ تم إنشاء الصور بنجاح!")
